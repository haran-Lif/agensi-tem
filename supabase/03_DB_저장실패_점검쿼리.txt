DB 저장 실패 점검 순서
======================

1. 먼저 014_db_save_reliability.sql 전체를 실행합니다.
2. 아래 확인 쿼리를 DBeaver에서 각각 실행합니다.
3. HTML을 새로고침하고 다시 로그인한 뒤 저장합니다.
4. 화면 우측 상단의 DB 상태에 실패 테이블명이 표시되면 해당 메시지 전체를 확인합니다.


[A. 필수 칼럼 누락 확인]
결과가 0행이면 정상입니다.

with required(table_name, column_name) as (
  values
    ('projects', 'unit_price'),
    ('projects', 'assigned_team_id'),
    ('projects', 'sla_days'),
    ('projects', 'parent_project_id'),
    ('teams', 'base_group'),
    ('teams', 'subgroup_name'),
    ('teams', 'subgroups'),
    ('employees', 'subgroup_name'),
    ('employees', 'resignation_reason'),
    ('employees', 'reduced_work_yn'),
    ('employees', 'reduced_work_start_date'),
    ('employees', 'reduced_work_end_date'),
    ('employees', 'reduced_work_start_time'),
    ('employees', 'reduced_work_end_time'),
    ('employees', 'reduced_daily_hours'),
    ('task_history', 'effort_buffer_pct'),
    ('task_history', 'photo_type'),
    ('output_history', 'owner_employee_id'),
    ('output_rules', 'sla_days'),
    ('output_rules', 'sla_enabled')
)
select r.table_name, r.column_name as missing_column
from required r
left join information_schema.columns c
  on c.table_schema = 'public'
 and c.table_name = r.table_name
 and c.column_name = r.column_name
where c.column_name is null
order by r.table_name, r.column_name;


[B. 업무 템플릿과 변경 이력 RLS 정책 확인]
output_rules_admin_write, project_changes_admin_write가 각각 1행이면 정상입니다.

select schemaname, tablename, policyname, cmd, roles, qual, with_check
from pg_policies
where schemaname = 'public'
  and tablename in ('output_rules', 'project_changes')
  and policyname in ('output_rules_admin_write', 'project_changes_admin_write')
order by tablename, policyname;

[D. 환산 기준 연결 오류 확인]
결과가 0행이면 정상입니다.

select
  r.rule_id,
  r.project_id,
  r.deliverable_name,
  r.conversion_target_rule_id
from public.output_rules r
left join public.output_rules target
  on target.rule_id = r.conversion_target_rule_id
where r.conversion_target_rule_id is not null
  and target.rule_id is null
order by r.project_id, r.deliverable_name;


[E. 같은 프로젝트 안의 산출물 명칭 중복 확인]
결과가 있으면 해당 항목을 업무 템플릿에서 하나로 정리합니다.

select project_id, trim(deliverable_name) as deliverable_name, count(*) as duplicate_count
from public.output_rules
group by project_id, trim(deliverable_name)
having count(*) > 1
order by duplicate_count desc, project_id, deliverable_name;

[G. 계정-직원-팀-프로젝트 자동 연결 확인]
015_team_project_account_flow.sql 적용 후 사용합니다.

select
  ap.login_id,
  ap.display_name as employee_name,
  ap.team_name,
  t.team_id,
  pr.project_id,
  pr.project_name
from public.app_profiles ap
left join public.teams t
  on t.team_name = ap.team_name
left join public.projects pr
  on pr.assigned_team_id = t.team_id
where ap.status = '사용'
order by ap.login_id, pr.project_name;


[H. employees 저장 실패 원인 확인]
아래 첫 쿼리에서 employees_admin_write가 1행이어야 합니다.
두 번째 쿼리는 직원 테이블에 적용된 제약조건을 보여줍니다.

select schemaname, tablename, policyname, cmd, roles, qual, with_check
from pg_policies
where schemaname = 'public'
  and tablename = 'employees'
order by policyname;

select
  con.conname as constraint_name,
  pg_get_constraintdef(con.oid) as definition
from pg_constraint con
where con.conrelid = 'public.employees'::regclass
order by con.conname;

-- 직원과 연결되지 않은 팀/직급 ID가 있으면 결과가 표시됩니다.
-- 결과가 0행이면 정상입니다.
select
  e.employee_id,
  e.employee_name,
  e.team_id,
  e.position_id,
  case
    when t.team_id is null then '팀 연결 누락'
    when p.position_id is null then '직급 연결 누락'
    else '정상'
  end as issue
from public.employees e
left join public.teams t on t.team_id = e.team_id
left join public.positions p on p.position_id = e.position_id
where t.team_id is null
   or p.position_id is null
order by e.employee_name;
