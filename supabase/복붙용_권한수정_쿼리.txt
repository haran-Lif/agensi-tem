[1] 새 권한 operator_output_admin 허용

begin;

alter table public.app_profiles
  drop constraint if exists app_profiles_role_check;

alter table public.app_profiles
  add constraint app_profiles_role_check
  check (role in (
    'admin',
    'operator_all',
    'operator_no_profit',
    'operator_output_admin',
    'portal',
    'worker'
  ));

comment on constraint app_profiles_role_check on public.app_profiles is
  '앱 로그인 권한. operator_output_admin은 운영자 메뉴와 관리자 업무 템플릿만 허용';

commit;


[2] worker 계정 세팅

update public.app_profiles
set role = 'operator_output_admin',
    display_name = 'test',
    updated_at = now()
where login_id = 'worker';

delete from public.app_user_projects
where user_id = (
  select user_id
  from public.app_profiles
  where login_id = 'worker'
);

insert into public.app_user_projects (user_id, project_name)
select user_id, 'test1'
from public.app_profiles
where login_id = 'worker'
on conflict do nothing;


[3] 적용 확인

select
  p.login_id,
  p.display_name,
  p.role,
  p.status,
  string_agg(aup.project_name, ', ' order by aup.project_name) as projects
from public.app_profiles p
left join public.app_user_projects aup
  on aup.user_id = p.user_id
where p.login_id = 'worker'
group by p.login_id, p.display_name, p.role, p.status;
